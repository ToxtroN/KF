# Собирает веб-версию игры в Godot и выкладывает её на GitHub Pages
# после каждой загрузки файлов в ветку main.
name: Игра на GitHub Pages

on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: true

env:
  GODOT_VERSION: "4.3"

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Скачать Godot и шаблоны для веба
        run: |
          base=https://github.com/godotengine/godot/releases/download/${GODOT_VERSION}-stable
          wget -q $base/Godot_v${GODOT_VERSION}-stable_linux.x86_64.zip
          unzip -q Godot_v${GODOT_VERSION}-stable_linux.x86_64.zip
          mv Godot_v${GODOT_VERSION}-stable_linux.x86_64 godot
          wget -q $base/Godot_v${GODOT_VERSION}-stable_export_templates.tpz
          unzip -q Godot_v${GODOT_VERSION}-stable_export_templates.tpz 'templates/web_nothreads_release.zip' 'templates/version.txt'
          mkdir -p ~/.local/share/godot/export_templates/${GODOT_VERSION}.stable
          mv templates/* ~/.local/share/godot/export_templates/${GODOT_VERSION}.stable/
          rm -f Godot_v${GODOT_VERSION}-stable_export_templates.tpz

      - name: Собрать веб-версию
        run: |
          mkdir -p build
          ./godot --headless --import || true
          ./godot --headless --export-release "Web" build/index.html
          ls -la build
          test -f build/index.wasm

      - uses: actions/upload-pages-artifact@v3
        with:
          path: build

  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - id: deployment
        uses: actions/deploy-pages@v4
